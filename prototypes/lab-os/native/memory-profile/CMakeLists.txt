cmake_minimum_required(VERSION 3.14)
project(labos_memory_profile C ASM)
set(CMAKE_C_STANDARD 11)
get_filename_component(LAB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(WAMR_ROOT_DIR "${LAB_ROOT}/downloads/os-research/wamr-upstream")
set(WAMR_BUILD_PLATFORM "darwin")
set(WAMR_BUILD_TARGET "AARCH64")
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
# Exercise the actual S3 mapping adapter, not Darwin's virtual mmap backend.
list(FILTER WAMR_RUNTIME_LIB_SOURCE EXCLUDE REGEX "/posix_memmap\\.c$")
include("${LAB_ROOT}/firmware/esp-idf/lab-os-core/components/wamr/ems_accounting.cmake")
lab_wamr_ems_accounting("${WAMR_ROOT_DIR}" "${CMAKE_BINARY_DIR}/generated" LAB_EMS_SOURCE)
list(FILTER WAMR_RUNTIME_LIB_SOURCE EXCLUDE REGEX "/ems/ems_alloc\\.c$")
list(APPEND WAMR_RUNTIME_LIB_SOURCE "${LAB_EMS_SOURCE}")
include_directories("${WAMR_ROOT_DIR}/core/shared/mem-alloc/ems")
add_library(labos-memory-runtime STATIC ${WAMR_RUNTIME_LIB_SOURCE}
  "${LAB_ROOT}/firmware/esp-idf/lab-os-core/components/wamr/platform_memory.c")
include_directories("${LAB_ROOT}/prototypes/lab-os/native")
add_executable(labos-memory-probe probe.c "${LAB_ROOT}/prototypes/lab-os/native/guest_vm.c")
target_compile_options(labos-memory-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(labos-memory-probe PRIVATE labos-memory-runtime pthread m)
